ТемаСцена.Метка (ThemeScene.Tag) Свойство

Использование

Чтение и запись.

Значение

Тип: Произвольный.

Возвращает или задает объект, содержащий дополнительные данные.

Примечание

Пример

Структура1 = Новый Структура("Буратино", "Деревянная кукла, наивная, добродушная и озорная.");
Структура1.Вставить("Карло", "Шарманщик, который вырезал Буратино из полена.");
Структура1.Вставить("Дуремар", "Продавец лечебных пиявок и друг Карабаса.");
Структура1.Вставить("Тортила", "Добрая черепаха, которая подарила Буратино золотой ключик.");
ТемаСцена1.Метка = Структура1;
Полный пример кода 



Перем ИФ, ТемаСцена1, ПутьШрифта;

Процедура ОдноразовыйТаймер() Экспорт
	Сообщить("ОдноразовыйТаймер " + ИФ.ТекущаяДатаСМиллисекундами());
	
	Сообщить("ТемаСцена1.Метка.Тортила = " + ТемаСцена1.Метка.Тортила);
	
	ИФ.СообщитьЕслиЛожь(ТемаСцена1.Метка.Тортила = "Добрая черепаха, которая подарила Буратино золотой ключик."	
	, "ТемаСцена.Метка (ThemeScene.Tag) Свойство");
	ИФ.Завершить();

ТемаСцена1.УстановитьЦвет(Иф.ИмяСвойстваТемыОкно.ЦветЗаголовка, Иф.Цвет.Красный);
ТемаСцена1.УстановитьКонстанту(Иф.ИмяСвойстваТемыОкно.ВысотаЗаголовка, 80);
ТемаСцена1.УстановитьКонстанту(Иф.ИмяСвойстваТемыОкно.СмещениеКнопкиЗакрытияИгрек, 60);
ТекстураИзИзображения = ИФ.ТекстураИзИзображения(ОбъединитьПути(ТекущийСценарий().Каталог, "arrowLeft.png"));

Окно2.Правее(Окно1, 20);
