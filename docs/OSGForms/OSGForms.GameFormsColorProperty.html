ИгровыеФормы.Цвет (GameForms.Color) Свойство

Использование

Только чтение.

Значение

Тип: Цвет (Color).

Возвращает экземпляр класса Цвет (Color).

Примечание

Пример

Цвет1 = ИФ.Цвет;
Полный пример кода 



Перем ИФ;

Процедура ОдноразовыйТаймер() Экспорт
	Цвет1 = ИФ.Цвет;
	Сообщить("Цвет1 = " + Цвет1);
	// Цвет через rgba
	Цвет2 = ИФ.Цвет(127, 255, 212, 110);
	Сообщить("Цвет2 = " + Цвет2);
	// Цвет через rgb
	Цвет3 = ИФ.Цвет(0,0,0);
	Сообщить("Цвет3 = " + Цвет3);
	// Цвет через свойство
	Цвет4 = ИФ.Цвет.Синий;
	Сообщить("Цвет4 = " + Цвет4);
	// Цвет через имя свойства
	Цвет5 = ИФ.Цвет["Синий"];
	Сообщить("Цвет5 = " + Цвет5);
	// Получить имя цвета
	Сообщить("Цвет5.Имя = " + Цвет5.Имя);
	Сообщить("Цвет5.Name = " + Цвет5.Name);
	// Приведение к строке
	Сообщить("Цвет5.ВСтроку() = " + Цвет5.ВСтроку());
	// Цвет из десятичного числа
	Цвет6 = ИФ.Цвет(11529966);
	Сообщить("Цвет6 = " + Цвет6);
	// Цвет из hex
	Цвет7 = ИФ.Цвет("#f5f5dCFF");
	Сообщить("Цвет7 = " + Цвет7);
	Сообщить("Цвет2.Получить16() = " + Цвет2.Получить16());

	// Перечень имен цветов
	М = ИФ.МассивИменЦветов();
	Сообщить("М = " + М.Количество());
	Для А = 0 По М.ВГраница() Цикл
		Сообщить("Цвет = " + М[А]);
	КонецЦикла;
	
	Сообщить("Строка(ТипЗнч(Цвет1)) = " + Строка(ТипЗнч(Цвет1)));
	ИФ.СообщитьЕслиЛожь(Строка(ТипЗнч(Цвет1)) = "ИфЦвет", "ИгровыеФормы.Цвет (GameForms.Color) Свойство");

	ИФ.Завершить();
